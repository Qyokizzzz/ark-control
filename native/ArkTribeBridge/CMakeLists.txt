cmake_minimum_required(VERSION 3.24)
project(ArkTribeBridge LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ASE ArkApi DLL requires Windows x64")
endif()
set(ARKAPI_SDK "" CACHE PATH "Path containing ArkApi headers and ArkApi.lib")
find_path(ARKAPI_INCLUDE API/ARK/Ark.h HINTS "${ARKAPI_SDK}" "${ARKAPI_SDK}/include" "${ARKAPI_SDK}/version/Core/Public" REQUIRED)
find_library(ARKAPI_LIBRARY ArkApi HINTS "${ARKAPI_SDK}" "${ARKAPI_SDK}/lib" "${ARKAPI_SDK}/out_lib" REQUIRED)
add_library(ArkTribeBridge SHARED Plugin.cpp)
target_compile_features(ArkTribeBridge PRIVATE cxx_std_17)
target_include_directories(ArkTribeBridge PRIVATE "${ARKAPI_INCLUDE}")
target_link_libraries(ArkTribeBridge PRIVATE "${ARKAPI_LIBRARY}")
target_compile_definitions(ArkTribeBridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_property(TARGET ArkTribeBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
